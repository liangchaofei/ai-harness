第 40 篇：比较两次评估，定位哪项任务退化了 ​

第 39 篇留下两份发票与路径任务的报告。基线通过 2 项，候选通过 1 项。整批比例能告诉我们有变化，排查时还要知道是哪一项，以及候选少做了什么。本篇实现 --compare-eval，按用例 ID 对齐报告，将接受变未接受的项标为回归，并为完整比较返回可供脚本使用的退出码。

第一次阅读，先跟以下路径任务，再到第一步的 compareCase 看交付和工具计数如何分别比较。已有第 39 篇的两个 report.json 时，可以直接用文末命令。图解可以点击查看原图。

本篇技术方案 ​

输入顺序是基线、候选，差值一律按「候选减基线」计算。先检查两份报告各自成立，再用稳定 ID 匹配同一任务。相同清单路径、相同用例集合、相同任务路径和完整批次是本篇的可比条件。结果仍要由操作者用相同版本的材料与验收生成。第 39 篇报告只记录路径，相同路径无法证明文件内容没变。

实施前，先走完一次具体过程 ​

  1. 操作者将第 39 篇的 baseline/report.json 和 skip-route-check/report.json 交给 CLI。比较器只读两个文件，分别复算 accepted 与 acceptanceRate，检查它们是否符合逐项结果。
  2. 基线的 invoice 按 ID 找到候选的 invoice，两次都 accepted，因此标为 unchanged。候选报告里的数组即使顺序倒过来，查找关系也一样。
  3. 基线的 route 为 accepted，候选为 notAccepted，因此是 regression。两份轨迹的停止状态都为 answered，候选的工具处理数却从 2 变为 1。把两者一起显示，读者可以回到轨迹核对少了验收。
  4. 整批接受率差值为 -0.5，对应少 50 个百分点。CLI 退出 1。若候选在路径项之前取消，比较器保留发票项的对照结果，路径项报 missingCase，整批差值留空并退出 2。

核心在 apps/coding-cli/src/eval/compare-reports.ts 的 compareEvalReports 和 compareCase。 前者决定哪两条记录能配对，后者根据交付结果分类。运行计数和耗时差值便于解释。一次短运行耗时更少，可能只是少跑了验收，因此门禁由交付退化决定。

Pi 的 buildEntryIndex 使用 Map 通过持久化记录的 ID 建索引，后续路径查找无需依赖数组位置。本篇把同一索引做法用于评估报告的用例 ID。 Pi 固定版本的 buildEntryIndex

第一步：读报告、按身份配对，再计算变化 ​

新建 apps/coding-cli/src/eval/compare-reports.ts。它接收两份报告的路径与取消信号，返回逐项对照和 CLI 退出码。输入数据由第 39 篇的任务集入口生成。

ts
// 【新增文件】按用例 ID 比较两份版本 1 的评估报告。
import { constants } from 'node:fs';
import { open } from 'node:fs/promises';
import { z } from 'zod';

const MAX_REPORT_BYTES = 256 * 1024;
const count = z.number().int().nonnegative();
const caseSchema = z.object({
  id: z.string().regex(/^[a-z][a-z0-9-]{0,39}$/),
  taskFile: z.string().min(1),
  outcome: z.enum(['accepted', 'notAccepted', 'infrastructureFailure', 'cancelled']),
  elapsedMs: count,
  metrics: z.object({
    status: z.enum(['answered', 'callLimit', 'truncated', 'emptyAnswer', 'failed', 'cancelled', 'timedOut', 'toolLimit']),
    modelCalls: count,
    toolCalls: count,
  }).nullable(),
});
const reportSchema = z.object({
  version: z.literal(1),
  manifest: z.string().min(1),
  planned: count.min(1).max(20),
  evaluated: count,
  accepted: count,
  acceptanceRate: z.number().min(0).max(1).nullable(),
  cancelled: z.boolean(),
  results: z.array(caseSchema).max(20),
});

/** 只读已有报告。先核对结构，再核对汇总能否由逐项结果重算。 */
async function readReport(path: string, signal: AbortSignal) {
  signal.throwIfAborted();
  const file = await open(path, constants.O_RDONLY | constants.O_NONBLOCK);
  try {
    if (!(await file.stat()).isFile()) {
      throw new Error('评估报告应为普通文件。');
    }
    const chunks: Buffer[] = [];
    for await (const chunk of file.createReadStream({
      end: MAX_REPORT_BYTES,
      autoClose: false,
      signal,
    })) {
      chunks.push(chunk);
    }
    const data = Buffer.concat(chunks);
    if (data.length > MAX_REPORT_BYTES) {
      throw new Error('评估报告超过 256 KiB。');
    }

    const report = reportSchema.parse(JSON.parse(data.toString('utf8')));
    const byId = new Map(report.results.map((item) => [item.id, item]));
    const accepted = report.results.filter((item) => item.outcome === 'accepted').length;
    const incomplete = report.cancelled || report.evaluated !== report.planned ||
      report.results.some((item) => item.outcome === 'infrastructureFailure' || item.outcome === 'cancelled');
    const acceptanceRate = incomplete ? null : accepted / report.planned;
    if (byId.size !== report.results.length || report.evaluated !== report.results.length ||
        report.evaluated > report.planned || report.accepted !== accepted ||
        report.acceptanceRate !== acceptanceRate ||
        (!report.cancelled && report.results.some((item) => item.outcome === 'cancelled'))) {
      throw new Error('评估报告汇总与逐项结果不一致。');
    }
    return { report, byId };
  } finally {
    await file.close();
  }
}

/** 交付变化单独分类。运行次数只在两边都有计数时才计算差值。 */
function compareCase(baseline: z.infer<typeof caseSchema>, candidate: z.infer<typeof caseSchema>) {
  if (baseline.taskFile !== candidate.taskFile) {
    return { change: 'incomparable' as const, reason: 'taskPathMismatch' };
  }
  const comparable = (outcome: string) => outcome === 'accepted' || outcome === 'notAccepted';
  if (!comparable(baseline.outcome) || !comparable(candidate.outcome)) {
    return { change: 'incomparable' as const, reason: 'missingOutcome' };
  }
  const change = baseline.outcome === candidate.outcome
    ? 'unchanged' as const
    : candidate.outcome === 'accepted' ? 'improvement' as const : 'regression' as const;

  return {
    change,
    baselineOutcome: baseline.outcome,
    candidateOutcome: candidate.outcome,
    elapsedDeltaMs: candidate.elapsedMs - baseline.elapsedMs,
    metricsDelta: baseline.metrics && candidate.metrics ? {
      baselineStatus: baseline.metrics.status,
      candidateStatus: candidate.metrics.status,
      modelCalls: candidate.metrics.modelCalls - baseline.metrics.modelCalls,
      toolCalls: candidate.metrics.toolCalls - baseline.metrics.toolCalls,
    } : null,
  };
}

/** 比较方向固定为基线到候选；缺少整批对照条件时返回 2。 */
export async function compareEvalReports(baselinePath: string, candidatePath: string, signal: AbortSignal) {
  const [baseline, candidate] = await Promise.all([
    readReport(baselinePath, signal),
    readReport(candidatePath, signal),
  ]);
  const sameManifest = baseline.report.manifest === candidate.report.manifest;
  const ids = new Set([...baseline.byId.keys(), ...candidate.byId.keys()]);
  const cases = [...ids].map((id) => {
    const before = baseline.byId.get(id);
    const after = candidate.byId.get(id);
    if (!sameManifest) {
      return { id, change: 'incomparable' as const, reason: 'manifestPathMismatch' };
    }
    if (!before || !after) {
      return { id, change: 'incomparable' as const, reason: 'missingCase' };
    }
    return { id, ...compareCase(before, after) };
  });

  const baselineAcceptanceRate = baseline.report.acceptanceRate;
  const candidateAcceptanceRate = candidate.report.acceptanceRate;
  const comparable = sameManifest && baselineAcceptanceRate !== null &&
    candidateAcceptanceRate !== null && cases.every((item) => item.change !== 'incomparable');
  const regressions = cases.filter((item) => item.change === 'regression').length;
  const improvements = cases.filter((item) => item.change === 'improvement').length;
  const acceptanceRateDelta = comparable
    ? candidateAcceptanceRate - baselineAcceptanceRate
    : null;
  return {
    report: {
      comparable,
      baselineAcceptanceRate,
      candidateAcceptanceRate,
      acceptanceRateDelta,
      regressions,
      improvements,
      cases,
    },
    exitCode: !comparable ? 2 : regressions > 0 ? 1 : 0,
  };
}

为什么要复算报告汇总 ​

在同文件的 readReport 中，文件只读打开。读取范围最多为 256 KiB 加一个用于检查超限的字节。随后 reportSchema 校验版本、结果字段与数值类型，然后执行下面的复算。

ts
const report = reportSchema.parse(JSON.parse(data.toString('utf8')));
const byId = new Map(report.results.map((item) => [item.id, item]));
const accepted = report.results.filter((item) => item.outcome === 'accepted').length;
const incomplete = report.cancelled || report.evaluated !== report.planned ||
  report.results.some((item) => item.outcome === 'infrastructureFailure' || item.outcome === 'cancelled');
const acceptanceRate = incomplete ? null : accepted / report.planned;
if (byId.size !== report.results.length || report.evaluated !== report.results.length ||
    report.evaluated > report.planned || report.accepted !== accepted ||
    report.acceptanceRate !== acceptanceRate ||
    (!report.cancelled && report.results.some((item) => item.outcome === 'cancelled'))) {
  throw new Error('评估报告汇总与逐项结果不一致。');
}
return { report, byId };

例如候选的结果是发票 accepted 和路径 notAccepted，所以 accepted 必须为 1。完整批次的接受率必须为 1 / 2。若有人只改汇总数字，没有对应的逐项结果，读报告时就会报错。重复 ID 也会触发失败，因为 Map 中同一个键只占一个位置。

取消或基础设施失败时，原任务集的接受率为 null。读取器允许这一结果，后面仍可对齐已完成的项。报告语法有效与整批可比较是两个独立判断。

为什么以两边 ID 的并集开始 ​

在同文件的 compareEvalReports 中，两份报告读完以后生成这段用例列表。

ts
const sameManifest = baseline.report.manifest === candidate.report.manifest;
const ids = new Set([...baseline.byId.keys(), ...candidate.byId.keys()]);
const cases = [...ids].map((id) => {
  const before = baseline.byId.get(id);
  const after = candidate.byId.get(id);
  if (!sameManifest) {
    return { id, change: 'incomparable' as const, reason: 'manifestPathMismatch' };
  }
  if (!before || !after) {
    return { id, change: 'incomparable' as const, reason: 'missingCase' };
  }
  return { id, ...compareCase(before, after) };
});

先遍历基线中的 ID，再补上只出现在候选里的 ID。于是无论哪边少一项，报告里都有一条记录说明缺失。报告中的清单路径不同会标为 manifestPathMismatch。同一 ID 仍会进入 compareCase 检查实际记录的任务路径，路径改了就记 taskPathMismatch。

交付退化和工具差值如何同时出现 ​

在同文件的 compareCase 中，先确认两边交付结果可比较，再根据候选的交付结果分类。

ts
function compareCase(baseline: z.infer<typeof caseSchema>, candidate: z.infer<typeof caseSchema>) {

以路径任务为例，基线是 accepted，候选是 notAccepted，这一步就能确定 regression。运行计数有独立条件：两边都有 metrics 时才计算差值。若有一边只有交付结果，metricsDelta 留空，regression 仍可成立。准备失败或取消时，缺少可用于本篇交付分类的结果，故返回 missingOutcome。

如何得到整批退出码 ​

在 compareEvalReports 最后，找到判断整批可比性的语句。

ts
const baselineAcceptanceRate = baseline.report.acceptanceRate;
const candidateAcceptanceRate = candidate.report.acceptanceRate;
const comparable = sameManifest && baselineAcceptanceRate !== null &&
  candidateAcceptanceRate !== null && cases.every((item) => item.change !== 'incomparable');
const regressions = cases.filter((item) => item.change === 'regression').length;
const improvements = cases.filter((item) => item.change === 'improvement').length;
const acceptanceRateDelta = comparable
  ? candidateAcceptanceRate - baselineAcceptanceRate
  : null;
return {
  report: {
    comparable,
    baselineAcceptanceRate,
    candidateAcceptanceRate,
    acceptanceRateDelta,
    regressions,
    improvements,
    cases,
  },
  exitCode: !comparable ? 2 : regressions > 0 ? 1 : 0,
};

comparable 还要求两份报告的接受率都有值，而且每个用例都可比较。两份报告不满足协定时，退出 2。完整比较只要发现一项退化就退出 1，即使另一项恰好改善、整体接受率没变。完整且没有退化时退出 0。这样脚本可以区分「看到了退化」和「证据不足，不能作门禁判断」。

// 【新增】回归比较只接收两份已写出的评估报告。
if (args[0] === '--compare-eval') {
  const [, baseline, candidate] = args;
  if (args.length !== 3 || !baseline?.trim() || baseline.startsWith('--') ||

// 只处理终端参数，不读取环境或启动任务。
import type { CheckDecision, CliOptions } from '../types/task';

export const helpText = `用法：ai-harness --task <任务.json> [--plan] [--read-only | --approve-tools] [--follow-up <问题>] [--session-log <文件>] [--trace-file <新文件>] [--checkpoint <文件> [--gate-check]]

ai-harness --inspect-trace <轨迹文件>
ai-harness --eval-suite <清单.json> --eval-output <新目录>
ai-harness --compare-eval <基线.json> <候选.json>
任务路径相对当前终端目录；配置内路径相对任务文件目录。
--plan 开启实验性计划；--follow-up 在首次回答后追加一条问题。
--read-only 只允许调查工具；--approve-tools 在修改或验收前逐次询问。
--gate-check 让保存的检查点等待人工批准，准备阶段暂不提供验收工具。
--trace-file 在新任务运行时保存结构化事件。
--help 只显示帮助。
退出码：0已接受，1未接受，2参数/准备/报告失败，130主动取消。`;

/** main传入argv，不接受未知、重复或缺值选项。 */
export function parseArgs(args: readonly string[]): CliOptions {
  if (args.length === 1 && args[0] === '--help') {
    return { help: true };
  }
  // 离线诊断只接受一个路径，独立于任务参数。
  if (args[0] === '--inspect-trace') {
    const path = args[1];
    if (args.length !== 2 || !path?.trim() || path.startsWith('--')) {
      throw new Error('轨迹诊断只接受一个文件路径。');
    }
    return { help: false, inspectTrace: path };
  }
  // 评估入口只接受清单路径和一个新目录。
  if (args[0] === '--eval-suite') {
    const [, suite, outputFlag, output] = args;
    if (args.length !== 4 || outputFlag !== '--eval-output' ||
        !suite?.trim() || suite.startsWith('--') ||
        !output?.trim() || output.startsWith('--')) {
      throw new Error('任务集评估需要清单和新输出目录。');
    }
    return { help: false, evalSuite: suite, evalOutput: output };
  }
  // 【新增】回归比较只接收两份已写出的评估报告。
  if (args[0] === '--compare-eval') {
    const [, baseline, candidate] = args;
    if (args.length !== 3 || !baseline?.trim() || baseline.startsWith('--') ||
        !candidate?.trim() || candidate.startsWith('--')) {
      throw new Error('评估比较按顺序接受基线和候选报告。');
    }
    return { help: false, compareEval: { baseline, candidate } };
  }
  let taskFile: string | undefined;
  let planning = false;
  let readOnly = false;
  let approveTools = false;
  let followUp: string | undefined;
  let sessionLog: string | undefined;
  let traceFile: string | undefined;
  let resumeLog: string | undefined;
  let checkpoint: string | undefined;
  let resumeCheck: string | undefined;
  let inspectCheck: string | undefined;
  // 保存时声明门禁，决定时另收路径和动作。
  let gateCheck = false;
  let decideCheck: { path: string; decision: CheckDecision } | undefined;
  for (let i = 0; i < args.length; i += 1) {
    const arg = args[i];
    if (arg === '--plan' && !planning) {
      planning = true;
    } else if (arg === '--read-only' && !readOnly) {
      readOnly = true;
    } else if (arg === '--approve-tools' && !approveTools) {
      approveTools = true;
    } else if (arg === '--follow-up' && followUp === undefined) {
      // 读取一条用户追问。引号由终端处理。
      const value = args[++i]?.trim();
      if (!value || value.startsWith('--')) {
        throw new Error('追问内容缺失。');
      }
      followUp = value;
    } else if (arg === '--session-log' && sessionLog === undefined) {
      // 文件路径相对启动 CLI 时的工作目录。
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('会话记录路径缺失。');
      }
      sessionLog = value;
    } else if (arg === '--trace-file' && traceFile === undefined) {
      // 路径由操作者提供，后续只允许新建文件。
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('诊断轨迹路径缺失。');
      }
      traceFile = value;
    } else if (arg === '--resume-log' && resumeLog === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('会话日志路径缺失。');
      }
      resumeLog = value;
    } else if (arg === '--checkpoint' && checkpoint === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('检查点路径缺失。');
      }
      checkpoint = value;
    } else if (arg === '--resume-check' && resumeCheck === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('恢复检查点路径缺失。');
      }
      resumeCheck = value;
    } else if (arg === '--inspect-check' && inspectCheck === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('检查点诊断路径缺失。');
      }
      inspectCheck = value;
    } else if (arg === '--gate-check' && !gateCheck) {
      gateCheck = true;
    } else if (arg === '--decide-check' && decideCheck === undefined) {
      const path = args[++i];
      const decision = args[++i];
      if (!path?.trim() || path.startsWith('--') || (decision !== 'approve' && decision !== 'deny')) {
        throw new Error('批准决定需要检查点路径以及 approve 或 deny。');
      }
      decideCheck = { path, decision };
    } else if (arg === '--task' && taskFile === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('任务路径缺失。');
      }
      taskFile = value;
    } else {
      throw new Error('参数无效。');
    }
  }
  if (resumeLog) {
    if (approveTools || readOnly || taskFile || sessionLog || traceFile || planning || checkpoint || resumeCheck || inspectCheck || gateCheck || decideCheck || !followUp) {
      throw new Error('只读接续需要且仅接受日志路径和一条追问。');
    }
    return { help: false, resumeLog, question: followUp };
  }
  if (!taskFile) {
    throw new Error('需要任务配置。');
  }
  // 这些入口不会启动任务模型循环。
  if (traceFile && (resumeCheck || inspectCheck || decideCheck)) {
    throw new Error('诊断轨迹只支持新任务运行。');
  }
  // 决定只写记录，不能与保存、运行或恢复混在一次调用中。
  if (gateCheck && !checkpoint) {
    throw new Error('--gate-check 需要 --checkpoint。');
  }
  if (decideCheck && (checkpoint || resumeCheck || inspectCheck || planning ||
      followUp || sessionLog || readOnly || approveTools)) {
    throw new Error('批准决定只接受任务配置、检查点路径和决定。');
  }
  if (approveTools && (readOnly || checkpoint || resumeCheck || inspectCheck)) {
    throw new Error('逐次批准不能与只读或检查点选项混用。');
  }
  if (readOnly && (planning || checkpoint || resumeCheck || inspectCheck)) {
    throw new Error('只读任务不能与计划或检查点选项混用。');
  }
  if (inspectCheck && (checkpoint || resumeCheck || followUp || planning || sessionLog)) {
    throw new Error('检查点诊断只接受任务配置与检查点路径。');
  }
  if ((checkpoint || resumeCheck) && (followUp || (resumeCheck && (checkpoint || planning || sessionLog)))) {
    throw new Error('检查点选项不能与当前参数组合。');
  }
  return {
    help: false,
    taskFile,
    planning,
    ...(readOnly ? { readOnly } : {}),
    ...(approveTools ? { approveTools } : {}),
    ...(followUp ? { followUp } : {}),
    ...(sessionLog ? { sessionLog } : {}),
    ...(traceFile ? { traceFile } : {}),
    ...(checkpoint ? { checkpoint } : {}),
    ...(gateCheck ? { gateCheck } : {}),
    ...(decideCheck ? { decideCheck } : {}),
    ...(resumeCheck ? { resumeCheck } : {}),
    ...(inspectCheck ? { inspectCheck } : {}),
  };
}

// 应用生命周期：参数→配置→副本→运行→报告→释放监听。
import type { RunInputOptions } from './types/task';
import type { ModelCall, RunEventListener } from '@ai-harness/runtime';
import { parseArgs, helpText } from './config/args';
import { loadTask } from './config/task';
import { createModelCall } from './model';
import { openInvoicePolicy } from './invoice-policy-mcp';
import { decidePendingCheck, inspectPendingCheck, resumePendingCheck, savePendingCheck } from './session/checkpoint';
import { prepareSession } from './session/prepare';
import { appendRun, createSessionLog } from './session/session-log';
import { resumeConversation } from './session/resume-conversation';
import { runSession } from './session/run';
import { reportSession } from './report/report';
import { printProgress, printBlock } from './report/terminal';
import { openRunTrace } from './report/trace';
import { inspectRunTrace } from './report/inspect-trace';
import { runEvalSuite } from './eval/run-suite';
import { compareEvalReports } from './eval/compare-reports';
import type { CliSession } from './types/session';

/** CLI 接收退出码。观察入口可注入模型，并开启全量历史对照。 */
export async function runCli(
  args: readonly string[],
  createCallModel: () => ModelCall = createModelCall,
  fullHistory = false,
): Promise<number> {
  const controller = new AbortController();
  const onInterrupt = () => {
    if (!controller.signal.aborted) {
      console.log('请求取消，等待当前协作操作退出。');
      controller.abort();
    }
  };
  let session: CliSession | undefined;
  let invoicePolicy: Awaited<ReturnType<typeof openInvoicePolicy>> | undefined;
  let trace: Awaited<ReturnType<typeof openRunTrace>> | undefined;
  let phase = '参数';
  process.on('SIGINT', onInterrupt);

  try {
    const options = parseArgs(args);
    if (options.help) {
      console.log(helpText);
      return 0;
    }

    // 【新增】比较既有报告，直接返回回归门禁退出码。
    if ('compareEval' in options) {
      phase = '评估报告比较';
      const { baseline, candidate } = options.compareEval;
      const { report, exitCode } = await compareEvalReports(baseline, candidate, controller.signal);
      printBlock('评估报告比较', report);
      return exitCode;
    }

    // 批次逐项重用单任务入口，每项准备新的副本与运行状态。
    if ('evalSuite' in options) {
      phase = '任务集评估';
      const { report, exitCode } = await runEvalSuite(
        options.evalSuite,
        options.evalOutput,
        (taskArgs) => runCli(taskArgs, createCallModel, fullHistory),
        controller.signal,
      );
      printBlock('任务集结果', report);
      return exitCode;
    }

    // 离线诊断先于模型、任务和工作副本装配。
    if ('inspectTrace' in options) {
      phase = '离线轨迹诊断';
      printBlock('离线轨迹诊断', await inspectRunTrace(options.inspectTrace, controller.signal));
      return 0;
    }

    // 只读接续先于任务配置和副本准备分流。
    if ('resumeLog' in options) {
      phase = '会话接续';
      const result = await resumeConversation(
        options.resumeLog,
        options.question,
        createCallModel,
        controller.signal,
      );
      printBlock('接续停止', { status: result.status });
      if (result.status === 'answered') {
        printBlock('接续回答', result.answer.content);
        return 0;
      }
      return result.status === 'cancelled' ? 130 : 1;
    }

    phase = '任务配置';
    const task = await loadTask(options.taskFile, controller.signal);

    // 批准命令只保存决定，之后由独立的恢复调用执行。
    if (options.decideCheck) {
      phase = '记录验收决定';
      const { path, decision } = options.decideCheck;
      const report = await decidePendingCheck(path, task, decision, controller.signal);
      printBlock('验收决定', report);
      return 0;
    }

    if (options.inspectCheck) {
      phase = '检查点诊断';
      const report = await inspectPendingCheck(options.inspectCheck, task, controller.signal);
      printBlock('检查点诊断', report);
      return 0;
    }

    if (options.resumeCheck) {
      phase = '检查点恢复';
      const result = await resumePendingCheck(options.resumeCheck, task, controller.signal);
      if (!('passed' in result)) {
        printBlock('验收未启动', result);
        return 1;
      }
      printBlock('恢复验收', result);
      return result.passed ? 0 : 1;
    }

    phase = '模型配置';
    const callModel = createCallModel();

    phase = '副本准备';
    session = await prepareSession(task, options.planning, controller.signal);
    printBlock('工作副本', session.root);

    // 先独占创建文件，防止两次 CLI 运行混写。
    let logPath: string | undefined;
    if (options.sessionLog) {
      phase = '会话记录';
      logPath = await createSessionLog(options.sessionLog, session.root);
    }

    // 准备模型循环前发现规则工具，首段和追问共用连接。
    if (task.extension === 'invoice-policy-mcp') {
      phase = '发票规则服务连接';
      invoicePolicy = await openInvoicePolicy(controller.signal);
    }

    // 开始循环前独占创建文件，首段和追问共用句柄。
    if (options.traceFile) {
      phase = '诊断轨迹';
      trace = await openRunTrace(options.traceFile);
    }

    phase = '运行与报告';
    const onEvent: RunEventListener = async (event) => {
      printProgress(event);
      if (
        event.type === 'tool_end' &&
        event.toolName === 'update_plan' &&
        !event.isError &&
        session?.plan
      ) {
        printBlock('任务计划（模型声明）', session.plan.read());
      }
      await trace?.append(event);
    };

    const runOptions: RunInputOptions = {
      fullHistory,
      ...(invoicePolicy ? { invoicePolicyTool: invoicePolicy.tool } : {}),
      ...(options.readOnly ? { readOnly: true } : {}),
      ...(options.approveTools ? { approveTools: true } : {}),
      ...(options.gateCheck ? { deferCheck: true } : {}),
    };
    const first = await runSession(task, session, callModel, controller.signal, onEvent, runOptions);
    // 首段落盘之后才启动追问。
    if (logPath) {
      phase = '会话记录';
      await appendRun(logPath, first, 0);
      phase = '运行与报告';
    }
    let result = first;

    // 沿用已经准备的副本和工具，只给 Runtime 一个新的历史起点。
    if (options.followUp && first.status === 'answered' && !controller.signal.aborted) {
      printBlock('首段停止', {
        status: first.status,
        modelCalls: first.modelCalls,
        toolCalls: first.toolCalls,
        runId: first.runId,
      });
      const nextInput: RunInputOptions = {
        messages: [...first.messages, { role: 'user', content: options.followUp }],
        ...runOptions,
      };
      result = await runSession(task, session, callModel, controller.signal, onEvent, nextInput);
      if (logPath) {
        phase = '会话记录';
        await appendRun(logPath, result, first.messages.length);
        phase = '运行与报告';
      }
    }

    if (options.checkpoint) {
      phase = '保存检查点';
      await savePendingCheck(
        options.checkpoint,
        task,
        session,
        result,
        controller.signal,
        options.gateCheck,
      );
      printBlock('检查点已保存', options.checkpoint);
      return 0;
    }

    return await reportSession(task, session, result);
  } catch {
    if (controller.signal.aborted) {
      return 130;
    }

    // 原异常留在调试器，不打印凭据、响应正文或整个请求。
    console.error(`${phase}失败；请检查配置或在对应函数断点查看原异常。`);
    return 2;
  } finally {
    if (trace) {
      await trace.close().catch(() => console.error('关闭诊断轨迹失败。'));
    }
    // 正常退出、异常和取消都要关闭本任务持有的连接。
    if (invoicePolicy) {
      await invoicePolicy.close().catch(() => console.error('关闭发票规则服务连接失败。'));
    }
    process.off('SIGINT', onInterrupt);
    if (session) {
      console.log('工作副本保留在：', session.root);
    }
  }
}
