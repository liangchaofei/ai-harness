本页内容

24 把已验收的经验留给下一个任务 ​

第 22 篇能读现成笔记。这一篇从空笔记开始。第一次路径修复验收通过后，CLI 记录根路径用例的预期结果。第二次启动同类任务时，CLI 从磁盘取出这条记录，交给模型当调查线索。

本篇在 Agent 的什么位置 ​

这是第四阶段「长任务的信息管理」的最后一篇。第 19 至 21 篇在任务运行中整理本轮输入。第 24 篇接在任务收尾，由 CLI 从当前验收中提取线索并保存。下一次任务启动时，第 22 篇的读取逻辑再把这条线索带入模型输入。

本篇技术方案 ​

实施前，先走完一次具体过程 ​

  1. 操作者在任务配置中指定笔记文件、主题 route-repair 和要记录的验收用例 /。首轮读取空笔记，模型照常调查并修复。
  2. 任务结束后，CLI 重新审阅当前副本。已有交付判断确认通过的验收对应当前文件，才允许尝试记笔记。
  3. 写入函数从验收输出里找到 input: "/"、expected: "/" 的通过项，生成一条带验收脚本名和内容标识的笔记。同一文件内用主题决定新增或更新。
  4. 下一次独立任务启动时，第 22 篇的读取入口从同一个文件找到这条笔记。旧笔记提供调查方向，本次是否通过仍由本次验收决定。

实现架构图 ​

验收标准。 第一次首轮无记忆，交付后文件多出一条；第二次使用新副本，首轮能读到它，重复保存同一证据保持一条。新证据可更新同主题，设为停用后读取和自动更新都跳过。应用负责存取，Runtime 继续管理当次运行历史。

本篇核心代码在哪 ​

看 apps/coding-cli/src/report/report.ts 的 reportSession。它在 acceptTask 得到 accepted 后，才把同一份 review 交给 saveAcceptedMemory。这几行决定什么时候允许记。

1.2 把读文件和选笔记分开 ​

打开 apps/coding-cli/src/context/task-memory.ts。先把 ../types/task 的类型导入改为 import type { MemoryNote, TaskMemory } from '../types/task';。然后在 isRecord 和 loadTaskMemory 之间加入下面这个函数。它只管读取和校验，选哪个主题仍由原来的函数决定。

/** 【新增】供读取与写入共用的文件校验，额外字段在写入时原样保留。 */
export async function readMemoryNotes(path: string, signal?: AbortSignal): Promise<MemoryNote[]> {
  const info = await stat(path);
  if (!info.isFile() || info.size > 8 * 1024) {
    throw new Error('任务记忆应为小型 JSON 文件。');
  }

  const raw: unknown = JSON.parse(await readFile(path, { encoding: 'utf8', signal }));
  if (!Array.isArray(raw)) {
    throw new Error('任务记忆应为记录数组。');
  }

  return raw.map((note: unknown) => {
    if (
      !isRecord(note) ||
      typeof note.topic !== 'string' ||
      typeof note.hint !== 'string' ||
      !note.hint.trim() ||
      note.hint.length > 240
    ) {
      throw new Error('记忆记录应包含 topic 和简短的 hint。');
    }
    return { ...note, topic: note.topic, hint: note.hint };
  });
}

ts
// 从操作者指定的资料中读取一条适用的历史线索。
import { readFile, stat } from 'node:fs/promises';
import { basename, resolve } from 'node:path';

import type { MemoryNote, TaskMemory } from '../types/task';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

/** 【新增】供读取与写入共用的文件校验，额外字段在写入时原样保留。 */
export async function readMemoryNotes(path: string, signal?: AbortSignal): Promise<MemoryNote[]> {
  const info = await stat(path);
  if (!info.isFile() || info.size > 8 * 1024) {
    throw new Error('任务记忆应为小型 JSON 文件。');
  }

  const raw: unknown = JSON.parse(await readFile(path, { encoding: 'utf8', signal }));
  if (!Array.isArray(raw)) {
    throw new Error('任务记忆应为记录数组。');
  }

  return raw.map((note: unknown) => {
    if (
      !isRecord(note) ||
      typeof note.topic !== 'string' ||
      typeof note.hint !== 'string' ||
      !note.hint.trim() ||
      note.hint.length > 240
    ) {
      throw new Error('记忆记录应包含 topic 和简短的 hint。');
    }
    return { ...note, topic: note.topic, hint: note.hint };
  });
}

/**
 * 读取任务配置指定的笔记，由 loadTask 在启动任务时调用。
 * @param selection 任务配置中尚未校验的 memory，包含 file 和 topic。
 * @param base 任务配置所在的目录，用来解析相对的 file。
 * @param signal 可选取消信号，传给笔记文件的读取操作。
 * @returns 匹配的一条记录。未配置或未命中时返回 undefined，资料无效或重复命中时抛错。
 */
export async function loadTaskMemory(
  selection: unknown,
  base: string,
  signal?: AbortSignal,
): Promise<TaskMemory | undefined> {
  if (selection === undefined) {
    return undefined;
  }

  if (!isRecord(selection) || typeof selection.file !== 'string' ||
      !selection.file.trim() || typeof selection.topic !== 'string' ||
      !selection.topic.trim() || selection.topic.length > 80) {
    throw new Error('任务记忆应指定 file 和 topic。');
  }

  const path = resolve(base, selection.file);
  const notes = await readMemoryNotes(path, signal);

  // 配置给出要找的主题，笔记给出线索正文，来源文件名由加载器补上。
  const topic = selection.topic.trim();
  const matches: TaskMemory[] = [];
  for (const note of notes) {
    // 【修改】停用的笔记仍留在文件里，加载时跳过。
    if (note.topic === topic && note.enabled !== false) {
      matches.push({ topic, source: basename(path), hint: note.hint });
    }
  }

  // 多条同主题记录需要操作者合并，避免用文件顺序决定哪条生效。
  if (matches.length > 1) {
    throw new Error('同一主题有多条记忆，请先合并。');
  }
  return matches[0];
}

第二步：从验收提取一条记录，再按主题保存 ​

新建 apps/coding-cli/src/context/save-memory.ts。它接收目标笔记和验收快照。先展开下方的完整实现写入新文件，再按下面的摘录走读。正式入口会在第三步接入。这里先只跟一条用例：操作者选的是 /，验收输出里要有同样的 input，且实际值与预期值相等。

完整实现 context/save-memory.ts
ts
// 【新增文件】从匹配当前副本的验收中提取一条线索，并按主题更新笔记。
import { mkdtemp, rename, rm, writeFile } from 'node:fs/promises';
import { basename, dirname, join } from 'node:path';
import type { CodeTaskReview } from '@ai-harness/coding-tools';
import type { MemoryCapture, MemoryNote } from '../types/task';
import { readMemoryNotes } from './task-memory';

/** 报告层读取此状态。unchanged 表示笔记内容和证据来源都相同。 */
export type MemorySaveStatus = 'no-evidence' | 'disabled' | 'unchanged' | 'saved';

function object(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

/** 报告确认交付后调用。验收输出提供用例，模型的最终回答不参与写入。 */
export async function saveAcceptedMemory(
  capture: MemoryCapture,
  review: CodeTaskReview,
  checkScript: string,
): Promise<MemorySaveStatus> {
  const check = review.lastCheck;
  if (!review.checkMatchesCurrent || check?.status !== 'passed' || check.exitCode !== 0) {
    return 'no-evidence';
  }

  const report: unknown = JSON.parse(check.stdout);
  if (!object(report) || report.passed !== true || !Array.isArray(report.cases)) {
    return 'no-evidence';
  }

  const sample = report.cases.find((item: unknown) =>
    object(item) && item.input === capture.caseInput && item.passed === true,
  );
  if (!object(sample) || typeof sample.expected !== 'string' || sample.expected.length > 64 ||
      sample.actual !== sample.expected) {
    return 'no-evidence';
  }

  const hint = `历史验收用例：${JSON.stringify(capture.caseInput)} 的预期结果是 ${JSON.stringify(sample.expected)}。请以当前规则和验收为准。`;
  if (hint.length > 240) {
    return 'no-evidence';
  }

  const notes = await readMemoryNotes(capture.path);
  const existing = notes.filter((note) => note.topic === capture.topic);
  if (existing.length > 1) {
    throw new Error('同一主题有多条记忆，请先合并。');
  }

  const previous = existing[0];
  if (previous?.enabled === false) {
    return 'disabled';
  }

  const evidence = { checkScript: basename(checkScript), revision: check.revision };
  if (previous?.hint === hint && object(previous.evidence) &&
      previous.evidence.checkScript === evidence.checkScript &&
      previous.evidence.revision === evidence.revision) {
    return 'unchanged';
  }

  const next: MemoryNote = { ...previous, topic: capture.topic, hint, evidence };
  const updated = previous
    ? notes.map((note) => note === previous ? next : note)
    : [...notes, next];
  const text = JSON.stringify(updated, null, 2) + '\n';
  if (Buffer.byteLength(text, 'utf8') > 8 * 1024) {
    throw new Error('写入后超过笔记文件大小上限。');
  }

  // 同目录暂存完整内容，写成功后再替换原文件。
  const staging = await mkdtemp(join(dirname(capture.path), '.memory-'));
  try {
    const pending = join(staging, 'notes.json');
    await writeFile(pending, text, { mode: 0o600 });
    await rename(pending, capture.path);
  } finally {
    await rm(staging, { recursive: true, force: true });
  }

  return 'saved';
}

);

如果找到 input: "/"、expected: "/"、actual: "/" 的通过项，后续代码就把预期值写进 hint。同时，evidence 保留验收脚本文件名和该次验收的 revision，用于核对来源。找不到合适的用例会返回 no-evidence，笔记保持原样。

为什么第二次保存不会再写一条 ​

在 apps/coding-cli/src/context/save-memory.ts 中看下面的摘录。它已包含在本步的完整实现里。

ts
  const previous = existing[0];
  if (previous?.enabled === false) {
    return 'disabled';
  }

  const evidence = { checkScript: basename(checkScript), revision: check.revision };
  if (previous?.hint === hint && object(previous.evidence) &&
      previous.evidence.checkScript === evidence.checkScript &&
      previous.evidence.revision === evidence.revision) {
    return 'unchanged';
  }

在 saveAcceptedMemory 中，首次运行没有旧条目，previous 是 undefined，两处提前返回都跳过。第二次运行找到同主题条目后，先检查它是否被停用。仍启用时，再比较提示、验收脚本名和内容标识。三项都相同就返回 unchanged，执行在打开暂存文件之前结束。如果示例换成 sample-v2，标识变化，代码继续走到下面的替换逻辑。

在同一文件的 saveAcceptedMemory 里，读完停用和重复判断后，再看下面这段。它已包含在完整实现里。

ts
  const next: MemoryNote = { ...previous, topic: capture.topic, hint, evidence };
  const updated = previous
    ? notes.map((note) => note === previous ? next : note)
    : [...notes, next];

首次空文件时，previous 为空，代码追加 next。第二次同主题命中后，若提示、脚本和内容标识都一致，会提前返回 unchanged。证据变化时，代码只替换原来那一条，其他主题照常保留。停用的笔记会先返回 disabled。

写文件时先在目标文件所在目录暂存完整 JSON，成功后再替换原文件。这样写到一半失败时，原笔记仍可读。本章的示例按顺序运行；多个进程同时更新同一文件需要额外的并发控制。

新建 examples/memory-writeback/inspect-memory.ts。它把三份人工验收依次交给刚写好的函数，最后用原来的读取函数核对内容。

完整文件 config/task.ts
ts
// 把操作者的可信任务配置转换为准备副本所需数据。
import { readFile, realpath, stat } from 'node:fs/promises';
import { dirname, isAbsolute, relative, resolve } from 'node:path';
import { loadTaskMemory } from '../context/task-memory';
import type { MemoryCapture, TaskConfig } from '../types/task';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

/** CLI路径按cwd解析；配置内source/checkScript按配置文件目录解析。 */
export async function loadTask(taskFile: string, signal?: AbortSignal): Promise<TaskConfig> {
  signal?.throwIfAborted();
  const configPath = await realpath(resolve(taskFile));
  const info = await stat(configPath);
  if (!info.isFile() || info.size > 16 * 1024) throw new Error('任务配置应为小型JSON文件。');
  const value: unknown = JSON.parse(await readFile(configPath, { encoding: 'utf8', signal }));
  const keys = ['source', 'prompt', 'writablePaths', 'checkScript'];
  // 保留四个必填字段，只额外允许可选的 memory。
  if (
    !isRecord(value) ||
    keys.some((key) => !Object.hasOwn(value, key)) ||
    Object.keys(value).some((key) => !keys.includes(key) && key !== 'memory')
  ) {
    throw new Error('任务字段不匹配。');
  }
  const { source, prompt, writablePaths, checkScript } = value;
  if (typeof source !== 'string' || !source.trim() || typeof checkScript !== 'string' || !checkScript.trim()
    || typeof prompt !== 'string' || !prompt.trim() || prompt.length > 4000
    || !Array.isArray(writablePaths) || writablePaths.length < 1 || writablePaths.length > 16
    || writablePaths.some((path) => typeof path !== 'string' || !path.trim())
    || new Set(writablePaths).size !== writablePaths.length) throw new Error('任务字段无效。');

  const base = dirname(configPath);
  const sourcePath = await realpath(resolve(base, source));
  const scriptPath = await realpath(resolve(base, checkScript));
  if (!(await stat(sourcePath)).isDirectory() || !(await stat(scriptPath)).isFile()
    || !scriptPath.endsWith('.mjs')) throw new Error('材料或验收文件无效。');
  const scriptRelative = relative(sourcePath, scriptPath);
  // 验收在来源之外，避免把它复制到模型处理的材料中；这不是OS隔离。
  if (scriptRelative === '' || (!isAbsolute(scriptRelative)
    && scriptRelative !== '..' && !scriptRelative.startsWith('../'))) throw new Error('验收应在材料目录之外。');
  // 只在加载任务时读取；模型调用期间复用选中的一条记录。
  const memory = await loadTaskMemory(value.memory, base, signal);
  // 【新增】读取笔记和授权写入是两项独立的配置决定。
  let memoryCapture: MemoryCapture | undefined;
  if (isRecord(value.memory) && Object.hasOwn(value.memory, 'captureCase')) {
    const { file, topic, captureCase } = value.memory;
    if (
      typeof file !== 'string' ||
      typeof topic !== 'string' ||
      typeof captureCase !== 'string' ||
      captureCase.length === 0 ||
      captureCase.length > 64
    ) {
      throw new Error('记忆写入需要指定可核对的验收用例。');
    }
    memoryCapture = {
      path: await realpath(resolve(base, file)),
      topic: topic.trim(),
      caseInput: captureCase,
    };
  }

  signal?.throwIfAborted();
  return {
    source: sourcePath,
    prompt: prompt.trim(),
    writablePaths: [...writablePaths],
    checkScript: scriptPath,
    ...(memory ? { memory } : {}),
    ...(memoryCapture ? { memoryCapture } : {}),
  };

}

完整文件 report/report.ts
ts
// 运行停止后从同会话取证，再按实际变化与验收判断交付。
import type { RunResult } from '@ai-harness/runtime';
import type { TaskConfig } from '../types/task';
import type { CliSession } from '../types/session';
import { saveAcceptedMemory } from '../context/save-memory';
import { acceptTask } from './accept';
import { printBlock } from './terminal';

/** main调用；inspect失败继续抛出，不能沿用旧报告或按模型总结接受。 */
export async function reportSession(
  task: TaskConfig,
  session: CliSession,
  result: RunResult,
): Promise<number> {
  // 从Runtime已保存的历史追踪请求与反馈，不把实时预览当完整记录。
  let responseNumber = 0;
  for (const message of result.messages) {
    if (message.role === 'assistant') {
      responseNumber += 1;
      printBlock(`模型响应 ${responseNumber}`, {
        stopReason: message.stopReason,
        content: message.content,
        toolCalls: message.toolCalls,
      });
    } else if (message.role === 'toolResult') {
      printBlock(`工具反馈 · ${message.toolName}`, {
        toolCallId: message.toolCallId,
        isError: message.isError,
        content: message.content,
      });
    }
  }

  printBlock('运行停止', {
    status: result.status,
    modelCalls: result.modelCalls,
    toolCalls: result.toolCalls,
    runId: result.runId,
    observerErrors: result.observerErrors,
  });
  if (result.status === 'failed') {
    printBlock('异常分类', { phase: result.phase, code: result.failure.code });
  }
  // 总结已包含在上面的完整模型消息中，不再单独重复打印。
  if (session.plan) {
    printBlock('任务计划（模型声明）', session.plan.read());
  }

  const review = await session.codeTask.inspect();
  for (const change of review.changes) {
    printBlock(`修改前 · ${change.path}`, change.before);
    printBlock(`当前内容 · ${change.path}`, change.after);
  }
  if (review.lastCheck) {
    printBlock('最近实际验收', review.lastCheck);
  }

  const accepted = acceptTask(result, review, task.writablePaths);
  printBlock('交付判断', {
    accepted,
    checkMatchesCurrent: review.checkMatchesCurrent,
    changedFiles: review.changes.map((change) => change.path),
  });
  console.log(
    accepted
      ? '已形成可审阅交付，尚未合入来源。'
      : '未接受：请查看停止原因、当前变化与实际验收。',
  );
  // 【新增】仅从已接受的交付提取笔记。记忆写入失败独立报告。
  if (accepted && task.memoryCapture) {
    try {
      const status = await saveAcceptedMemory(task.memoryCapture, review, task.checkScript);
      printBlock('长期记忆写入', { status });
    } catch {
      printBlock('长期记忆写入', { status: 'failed' });
    }
  }
  return result.status === 'cancelled' ? 130 : accepted ? 0 : 1;
}
