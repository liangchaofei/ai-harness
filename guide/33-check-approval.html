第 33 篇：让验收等待跨进程批准 ​

路径修复已经写进工作副本，但操作者还要先查看差异。此时可以退出 Coding CLI，稍后再对这一次验收作出决定。批准后，另一次 CLI 调用在原副本上运行验收。拒绝后，恢复调用能读到决定并停止。

第 32 篇的终端询问发生在工具调用内。操作者离开终端后仍需要审阅的场景，则需要把待执行的验收及其决定写到磁盘。这个入口位于应用层的任务交接处，准备修改时仍走现有 Agent Runtime。

图解可点击查看原图。首次阅读时，先跟一遍技术方案，再读第三步里 claimCheck 前面的判断，最后跑一次批准实验。看到 waiting 和 reused: false 后，再回头看决定文件如何与检查点配对。

本篇技术方案 ​

操作者创建检查点时选用 --gate-check。CLI 保存这项选择，后面启动的恢复进程直接从检查点读取它。这样，操作者执行普通的 --resume-check 时，仍会经过批准检查。

实施前，先走完一次具体过程 ​

  1. 用 route.json 准备 normalizePath 的修改。首个 CLI 进程仍可读取和修改获准文件，但暂时移除 run_check。模型返回后，CLI 把副本路径、内容指纹、任务指纹和门禁标记写进 route.pending.json。
  2. 操作者查看工作副本的改动，用 --inspect-check 看到 approval: 'waiting'。此时尝试恢复会得到「验收未启动」。磁盘上还没有 .started。
  3. 操作者另开一个进程，用 --decide-check route.pending.json approve。应用先核对副本和任务，再把检查点原文的 SHA-256 摘要和 approve 写进 route.pending.json.approval。摘要是对内容算出的固定长度标识，用来判断一份决定是否对应当前这份检查点。
  4. 再用 --resume-check。恢复进程重新核对副本和任务，读取匹配的批准，然后通过原有的 claimCheck 占用、执行验收并写回执。第二次恢复直接使用已有回执。若第三步写入 deny，第四步停在开始验收之前。

把三个文件分开，是因为它们问的是三个不同问题。

文件记录的事实写入方
route.pending.json.approval操作者对这份检查点批准或拒绝决定入口
route.pending.json.started有恢复进程已取得执行权claimCheck
route.pending.json.done一次实际验收完成后保存的结果恢复入口

首次批准实验的验收标准是：等批准时和记录批准后均无 .started，获准恢复后八项路径规则通过，再恢复得到 reused: true。拒绝实验必须有 .approval 且没有产生 .started 或 .done。把批准文件移到另一份检查点旁，也应在占用之前失败。

核心代码在 apps/coding-cli/src/session/checkpoint.ts 的 resumePendingCheck。 找到调用 claimCheck 的位置，再向上找读取批准的分支。谁能让执行往下走，取决于该分支读到的状态。另一个关键位置是 session/run.ts 的工具装配，负责把实际验收留给恢复入口。

这里处理可信本地任务和操作者控制的目录。同一检查点的并发恢复沿用文件独占机制。目录访问权限决定了谁能改动这些状态文件，摘要校验用来关联决定与请求。恢复范围是一次验收，后面的执行回执沿用第 28 至 30 篇。

Pi 的 SessionManager.appendCustomEntry 把扩展状态保存为会话条目，该条目由扩展使用且不进入模型上下文。本篇取其状态由应用持久化的设计思路，在自己的检查点旁保存独立决定文件。 Pi 固定版本的 session-manager.ts

--help 只显示帮助。

/** main传入argv，不接受未知、重复或缺值选项。 */
export function parseArgs(args: readonly string[]): CliOptions {
  if (args.length === 1 && args[0] === '--help') {
    return { help: true };
  }
  let taskFile: string | undefined;
  let planning = false;
  let readOnly = false;
  let approveTools = false;
  let followUp: string | undefined;
  let sessionLog: string | undefined;
  let resumeLog: string | undefined;
  let checkpoint: string | undefined;
  let resumeCheck: string | undefined;
  let inspectCheck: string | undefined;
  // 【新增】保存时声明门禁，决定时另收路径和动作。
  let gateCheck = false;
  let decideCheck: { path: string; decision: CheckDecision } | undefined;
  for (let i = 0; i < args.length; i += 1) {
    const arg = args[i];
    if (arg === '--plan' && !planning) {
      planning = true;
    } else if (arg === '--read-only' && !readOnly) {
      readOnly = true;
    } else if (arg === '--approve-tools' && !approveTools) {
      approveTools = true;
    } else if (arg === '--follow-up' && followUp === undefined) {
      // 读取一条用户追问。引号由终端处理。
      const value = args[++i]?.trim();
      if (!value || value.startsWith('--')) {
        throw new Error('追问内容缺失。');
      }
      followUp = value;
    } else if (arg === '--session-log' && sessionLog === undefined) {
      // 文件路径相对启动 CLI 时的工作目录。
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('会话记录路径缺失。');
      }
      sessionLog = value;
    } else if (arg === '--resume-log' && resumeLog === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('会话日志路径缺失。');
      }
      resumeLog = value;
    } else if (arg === '--checkpoint' && checkpoint === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('检查点路径缺失。');
      }
      checkpoint = value;
    } else if (arg === '--resume-check' && resumeCheck === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('恢复检查点路径缺失。');
      }
      resumeCheck = value;
    } else if (arg === '--inspect-check' && inspectCheck === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('检查点诊断路径缺失。');
      }
      inspectCheck = value;
    } else if (arg === '--gate-check' && !gateCheck) {
      gateCheck = true;
    } else if (arg === '--decide-check' && decideCheck === undefined) {
      const path = args[++i];
      const decision = args[++i];
      if (!path?.trim() || path.startsWith('--') || (decision !== 'approve' && decision !== 'deny')) {
        throw new Error('批准决定需要检查点路径以及 approve 或 deny。');
      }
      decideCheck = { path, decision };
    } else if (arg === '--task' && taskFile === undefined) {
      const value = args[++i];
      if (!value?.trim() || value.startsWith('--')) {
        throw new Error('任务路径缺失。');
      }
      taskFile = value;
    } else {
      throw new Error('参数无效。');
    }
  }
  if (resumeLog) {
    if (approveTools || readOnly || taskFile || sessionLog || planning || checkpoint || resumeCheck || inspectCheck || gateCheck || decideCheck || !followUp) {
      throw new Error('只读接续需要且仅接受日志路径和一条追问。');
    }
    return { help: false, resumeLog, question: followUp };
  }
  if (!taskFile) {
    throw new Error('需要任务配置。');
  }
  // 【新增】决定只写记录，不能与保存、运行或恢复混在一次调用中。
  if (gateCheck && !checkpoint) {
    throw new Error('--gate-check 需要 --checkpoint。');
  }
  if (decideCheck && (checkpoint || resumeCheck || inspectCheck || planning ||
      followUp || sessionLog || readOnly || approveTools)) {
    throw new Error('批准决定只接受任务配置、检查点路径和决定。');
  }
  if (approveTools && (readOnly || checkpoint || resumeCheck || inspectCheck)) {
    throw new Error('逐次批准不能与只读或检查点选项混用。');
  }
  if (readOnly && (planning || checkpoint || resumeCheck || inspectCheck)) {
    throw new Error('只读任务不能与计划或检查点选项混用。');
  }
  if (inspectCheck && (checkpoint || resumeCheck || followUp || planning || sessionLog)) {
    throw new Error('检查点诊断只接受任务配置与检查点路径。');
  }
  if ((checkpoint || resumeCheck) && (followUp || (resumeCheck && (checkpoint || planning || sessionLog)))) {
    throw new Error('检查点选项不能与当前参数组合。');
  }
  return {
    help: false,
    taskFile,
    planning,
    ...(readOnly ? { readOnly } : {}),
    ...(approveTools ? { approveTools } : {}),
    ...(followUp ? { followUp } : {}),
    ...(sessionLog ? { sessionLog } : {}),
    ...(checkpoint ? { checkpoint } : {}),
    ...(gateCheck ? { gateCheck } : {}),
    ...(decideCheck ? { decideCheck } : {}),
    ...(resumeCheck ? { resumeCheck } : {}),
    ...(inspectCheck ? { inspectCheck } : {}),
  };
}

// 【新增文件】在检查点旁记录操作者对这一次验收的决定。
import { readFile, writeFile } from 'node:fs/promises';
import type { CheckDecision } from '../types/task';

/** 当前检查点的批准状态。旧检查点由调用方单独处理。 */
export type CheckApproval = 'waiting' | 'approved' | 'denied';

/** 只接受与当前检查点原文匹配、状态合法的决定记录。 */
export async function readCheckApproval(
  path: string,
  checkpointKey: string,
  signal?: AbortSignal,
): Promise<CheckApproval> {
  let text: string;
  try {
    text = await readFile(path + '.approval', { encoding: 'utf8', signal });
  } catch (error) {
    if (error instanceof Error && 'code' in error && error.code === 'ENOENT') {
      return 'waiting';
    }
    throw error;
  }

  const record: unknown = JSON.parse(text);
  if (typeof record !== 'object' || record === null ||
      !('checkpointKey' in record) || record.checkpointKey !== checkpointKey ||
      !('decision' in record) || (record.decision !== 'approve' && record.decision !== 'deny')) {
    throw new Error('批准记录无效或与当前检查点不匹配。');
  }
  return record.decision === 'approve' ? 'approved' : 'denied';
}

/** 只允许第一份决定写入。重复提交不会覆盖已有决定。 */
export async function saveCheckApproval(
  path: string,
  checkpointKey: string,
  decision: CheckDecision,
  signal?: AbortSignal,
): Promise<void> {
  signal?.throwIfAborted();

    flag: 'wx',
    mode: 0o600,
  });
}

//保存并接续一个明确等待验收的工作副本。
import { createHash } from 'node:crypto';
import { readFile, realpath, stat, writeFile } from 'node:fs/promises';
import { isAbsolute, resolve } from 'node:path';
import type { RunResult } from '@ai-harness/runtime';
import type { CliSession } from '../types/session';
import type { CheckDecision, TaskConfig } from '../types/task';
import { readCheckApproval, saveCheckApproval } from './check-approval';
import { claimCheck, inspectCheckReceipt, saveCheckReceipt } from './check-receipt';
import { openTaskWorkspace } from './prepare';

/** 写入方创建，恢复方校验。记录一次待执行的验收及其适用内容。 */
interface PendingCheck {
  /** 检查点的磁盘格式版本。 */
  version: 1;
  /** 下一步固定为验收，恢复入口据此选择动作。 */
  next: 'check';
  /** 创建时确认的旧副本绝对路径。 */
  workspace: string;
  /** 写入检查点时，副本受跟踪文本范围的内容指纹。 */
  revision: string;
  /** 任务来源、可写路径、验收脚本位置和脚本正文的指纹。 */
  taskKey: string;
  /** 【新增】写入 true 才需要人工决定。旧检查点没有该字段。 */
  requiresApproval?: true;
}

function isPendingCheck(value: unknown): value is PendingCheck {
  if (typeof value !== 'object' || value === null) return false;
  const item = value as Record<string, unknown>;
  return item.version === 1 && item.next === 'check' &&
    typeof item.workspace === 'string' && isAbsolute(item.workspace) &&
    typeof item.revision === 'string' && /^[a-f0-9]{64}$/.test(item.revision) &&
    typeof item.taskKey === 'string' && /^[a-f0-9]{64}$/.test(item.taskKey) &&
    (item.requiresApproval === undefined || item.requiresApproval === true);
}

async function taskKey(task: TaskConfig, signal?: AbortSignal): Promise<string> {
  const script = await readFile(task.checkScript, { encoding: 'utf8', signal });
  return createHash('sha256').update(JSON.stringify([
    task.source,
    task.writablePaths,
    task.checkScript,
    script,
  ])).digest('hex');
}

/** 只在正常返回、已有允许改动且未请求验收时写下待验收检查点。 */
export async function savePendingCheck(
  path: string,
  task: TaskConfig,
  session: CliSession,
  result: RunResult,
  signal?: AbortSignal,
  requiresApproval = false,
): Promise<void> {
  const requestedCheck = result.messages.some((message) =>
    message.role === 'assistant' && message.toolCalls.some((call) => call.name === 'run_check'),
  );
  const review = await session.codeTask.inspect(signal);
  if (result.status !== 'answered' || requestedCheck || review.lastCheck ||
      review.changes.length === 0 ||
      review.changes.some((change) => !task.writablePaths.includes(change.path))) {
    throw new Error('只能保存已修改且尚未尝试验收的受控交接。');
  }

  const entry: PendingCheck = {
    version: 1,
    next: 'check',
    workspace: await realpath(session.root),
    revision: review.currentRevision,
    taskKey: await taskKey(task, signal),
    ...(requiresApproval ? { requiresApproval: true } : {}),
  };
  signal?.throwIfAborted();
  await writeFile(path, JSON.stringify(entry) + '\n', { flag: 'wx', mode: 0o600 });
}

async function openPendingCheck(path: string, task: TaskConfig, signal?: AbortSignal) {
  const checkpointPath = await realpath(resolve(path));
  const info = await stat(checkpointPath);
  if (!info.isFile() || info.size > 8 * 1024) {
    throw new Error('检查点文件无效或超限。');
  }
  const text = await readFile(checkpointPath, { encoding: 'utf8', signal });
  const entry: unknown = JSON.parse(text);
  if (Buffer.byteLength(text) > 8 * 1024 || !isPendingCheck(entry) ||
      entry.taskKey !== await taskKey(task, signal) ||
      await realpath(entry.workspace) !== entry.workspace) {
    throw new Error('检查点与当前任务不匹配。');
  }

  const session = await openTaskWorkspace(task, entry.workspace, signal);
  const before = await session.inspect(signal);
  if (before.currentRevision !== entry.revision) {
    throw new Error('副本内容与检查点不一致。');
  }
  const checkpointKey = createHash('sha256').update(text).digest('hex');
  return { checkpointPath, checkpointKey, entry, session };
}

/** 【新增】先核对任务和副本，再写入操作者对这一个待验收动作的决定。 */
export async function decidePendingCheck(
  path: string,
  task: TaskConfig,
  decision: CheckDecision,
  signal?: AbortSignal,
) {
  const { checkpointPath, checkpointKey, entry } = await openPendingCheck(path, task, signal);
  if (!entry.requiresApproval || (await inspectCheckReceipt(checkpointPath, checkpointKey)).status !== 'pending') {
    throw new Error('当前检查点没有可批准的待验收动作。');
  }
  await saveCheckApproval(checkpointPath, checkpointKey, decision, signal);
  return { workspace: entry.workspace, approval: decision === 'approve' ? 'approved' : 'denied' };
}

/** 操作者查看检查点的批准与执行证据。 */
export async function inspectPendingCheck(path: string, task: TaskConfig, signal?: AbortSignal) {
  const { checkpointPath, checkpointKey, entry } = await openPendingCheck(path, task, signal);
  const state = await inspectCheckReceipt(checkpointPath, checkpointKey);
  const approval = entry.requiresApproval
    ? await readCheckApproval(checkpointPath, checkpointKey, signal)
    : 'notRequired';
  return {
    workspace: entry.workspace,
    approval,
    status: state.status,
    ...(state.status === 'completed' ? { historicalPassed: state.receipt.passed } : {}),
  };
}

/** 先确认副本和任务，再占用检查点并执行一次原有验收。 */
export async function resumePendingCheck(path: string, task: TaskConfig, signal?: AbortSignal) {
  const { checkpointPath, checkpointKey, entry, session } = await openPendingCheck(path, task, signal);
  // 【新增】等待或拒绝时，在创建 .started 前退出。
  if (entry.requiresApproval) {
    const approval = await readCheckApproval(checkpointPath, checkpointKey, signal);
    if (approval !== 'approved') {
      return { workspace: entry.workspace, approval };
    }
  }
  const check = session.tools.find((tool) => tool.name === 'run_check');
  if (!check) throw new Error('当前任务缺少验收工具。');

  // 已有匹配回执时复用结果；已占用且无回执时由 claimCheck 拒绝继续。
  signal?.throwIfAborted();
  const receipt = await claimCheck(checkpointPath, checkpointKey);
  if (receipt) {
    return { workspace: entry.workspace, passed: receipt.passed, feedback: receipt.feedback, reused: true };
  }
  const feedback = await check.execute({}, signal);
  const after = await session.inspect(signal);
  const report: unknown = JSON.parse(after.lastCheck?.stdout ?? 'null');
  const passed = after.checkMatchesCurrent && after.currentRevision === entry.revision &&
    after.lastCheck?.status === 'passed' && after.lastCheck.exitCode === 0 &&
    typeof report === 'object' && report !== null && 'passed' in report && report.passed === true;
  // 先写回执，再向调用方返回结果。
  await saveCheckReceipt(checkpointPath, { checkpointKey, passed, feedback });
  return { workspace: entry.workspace, passed, feedback, reused: false };
}

// 应用生命周期：参数→配置→副本→运行→报告→释放监听。
import type { RunInputOptions } from './types/task';
import type { ModelCall, RunEventListener } from '@ai-harness/runtime';
import { parseArgs, helpText } from './config/args';
import { loadTask } from './config/task';
import { createModelCall } from './model';
import { decidePendingCheck, inspectPendingCheck, resumePendingCheck, savePendingCheck } from './session/checkpoint';
import { prepareSession } from './session/prepare';
import { appendRun, createSessionLog } from './session/session-log';
import { resumeConversation } from './session/resume-conversation';
import { runSession } from './session/run';
import { reportSession } from './report/report';
import { printProgress, printBlock } from './report/terminal';
import type { CliSession } from './types/session';

/** CLI 接收退出码。观察入口可注入模型，并开启全量历史对照。 */
export async function runCli(
  args: readonly string[],
  createCallModel: () => ModelCall = createModelCall,
  fullHistory = false,
): Promise<number> {
  const controller = new AbortController();
  const onInterrupt = () => {
    if (!controller.signal.aborted) {
      console.log('请求取消，等待当前协作操作退出。');
      controller.abort();
    }
  };
  let session: CliSession | undefined;
  let phase = '参数';
  process.on('SIGINT', onInterrupt);

  try {
    const options = parseArgs(args);
    if (options.help) {
      console.log(helpText);
      return 0;
    }

    // 只读接续先于任务配置和副本准备分流。
    if ('resumeLog' in options) {
      phase = '会话接续';
      const result = await resumeConversation(
        options.resumeLog,
        options.question,
        createCallModel,
        controller.signal,
      );
      printBlock('接续停止', { status: result.status });
      if (result.status === 'answered') {
        printBlock('接续回答', result.answer.content);
        return 0;
      }
      return result.status === 'cancelled' ? 130 : 1;
    }

    phase = '任务配置';
    const task = await loadTask(options.taskFile, controller.signal);

    // 【新增】批准命令只保存决定，之后由独立的恢复调用执行。
    if (options.decideCheck) {
      phase = '记录验收决定';
      const { path, decision } = options.decideCheck;
      const report = await decidePendingCheck(path, task, decision, controller.signal);
      printBlock('验收决定', report);
      return 0;
    }

    if (options.inspectCheck) {
      phase = '检查点诊断';
      const report = await inspectPendingCheck(options.inspectCheck, task, controller.signal);
      printBlock('检查点诊断', report);
      return 0;
    }

    if (options.resumeCheck) {
      phase = '检查点恢复';
      const result = await resumePendingCheck(options.resumeCheck, task, controller.signal);
      if (!('passed' in result)) {
        printBlock('验收未启动', result);
        return 1;
      }
      printBlock('恢复验收', result);
      return result.passed ? 0 : 1;
    }

    phase = '模型配置';
    const callModel = createCallModel();

    phase = '副本准备';
    session = await prepareSession(task, options.planning, controller.signal);
    printBlock('工作副本', session.root);

    // 先独占创建文件，防止两次 CLI 运行混写。
    let logPath: string | undefined;
    if (options.sessionLog) {
      phase = '会话记录';
      logPath = await createSessionLog(options.sessionLog, session.root);
    }

    phase = '运行与报告';
    const onEvent: RunEventListener = (event) => {
      printProgress(event);
      if (
        event.type === 'tool_end' &&
        event.toolName === 'update_plan' &&
        !event.isError &&
        session?.plan
      ) {
        printBlock('任务计划（模型声明）', session.plan.read());
      }
    };

    const runOptions: RunInputOptions = {
      fullHistory,
      ...(options.readOnly ? { readOnly: true } : {}),
      ...(options.approveTools ? { approveTools: true } : {}),
      ...(options.gateCheck ? { deferCheck: true } : {}),
    };
    const first = await runSession(task, session, callModel, controller.signal, onEvent, runOptions);
    // 首段落盘之后才启动追问。
    if (logPath) {
      phase = '会话记录';
      await appendRun(logPath, first, 0);
      phase = '运行与报告';
    }
    let result = first;

    // 沿用已经准备的副本和工具，只给 Runtime 一个新的历史起点。
    if (options.followUp && first.status === 'answered' && !controller.signal.aborted) {
      printBlock('首段停止', {
        status: first.status,
        modelCalls: first.modelCalls,
        toolCalls: first.toolCalls,
        runId: first.runId,
      });
      const nextInput: RunInputOptions = {
        messages: [...first.messages, { role: 'user', content: options.followUp }],
        ...runOptions,
      };
      result = await runSession(task, session, callModel, controller.signal, onEvent, nextInput);
      if (logPath) {
        phase = '会话记录';
        await appendRun(logPath, result, first.messages.length);
        phase = '运行与报告';
      }
    }

    if (options.checkpoint) {
      phase = '保存检查点';
      await savePendingCheck(
        options.checkpoint,
        task,
        session,
        result,
        controller.signal,
        options.gateCheck,
      );
      printBlock('检查点已保存', options.checkpoint);
      return 0;
    }

    return await reportSession(task, session, result);
  } catch {
    if (controller.signal.aborted) {
      return 130;
    }

    // 原异常留在调试器，不打印凭据、响应正文或整个请求。
    console.error(`${phase}失败；请检查配置或在对应函数断点查看原异常。`);
    return 2;
  } finally {
    process.off('SIGINT', onInterrupt);
    if (session) {
      console.log('工作副本保留在：', session.root);
    }
  }
}
